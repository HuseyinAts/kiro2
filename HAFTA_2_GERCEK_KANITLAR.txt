```typescript
const renderGraph = () => {
  const canvas = canvasRef.current;
  if (!canvas) return;

  const ctx = canvas.getContext('2d');
  if (!ctx) return;

  // Simple force-directed graph rendering
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // Draw nodes
  nodes.forEach(node => {
    ctx.beginPath();
    ctx.arc(node.x || 0, node.y || 0, 20, 0, 2 * Math.PI);
    ctx.fillStyle = node.status === 'weak' ? '#ef4444' : '#10b981';
    ctx.fill();
    ctx.fillStyle = '#000';
    ctx.fillText(node.label, (node.x || 0) + 25, (node.y || 0) + 5);
  });

  // Draw edges
  edges.forEach(edge => {
    const source = nodes.find(n => n.id === edge.source);
    const target = nodes.find(n => n.id === edge.target);
    if (source && target) {
      ctx.beginPath();
      ctx.moveTo(source.x || 0, source.y || 0);
      ctx.lineTo(target.x || 0, target.y || 0);
      ctx.strokeStyle = edge.relation === 'prerequisite' ? '#3b82f6' : '#9ca3af';
      ctx.stroke();
    }
  });
};
```

```python
            assert "question_id" in data
            assert "plagiarism_result" in data

    """Test CAT (Adaptive Testing) session flow"""
    async with AsyncClient(app=app, base_url="http://test") as client:
```
