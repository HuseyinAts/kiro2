# KIRO2 Secrets Patterns for git-filter-repo
# Format: literal:replacement (or regex:replacement)
# These patterns will be replaced with [REDACTED] in git history

# Anthropic API Keys
regex:sk-ant-api\d+-[A-Za-z0-9_-]{20,}==>[REDACTED_ANTHROPIC_KEY]

# OpenAI API Keys
regex:sk-proj-[A-Za-z0-9_-]{20,}==>[REDACTED_OPENAI_KEY]

# KIRO2 Database Passwords
literal:TeknoFest2025SecurePass==>[REDACTED_DB_PASSWORD]

# KIRO2 JWT Secrets
literal:teknofest-2025-super-secret-key-change-in-production==>[REDACTED_JWT_SECRET]

# Generic weak passwords found in code
literal:password='1470'==>password='[REDACTED]'
literal:password='admin123'==>password='[REDACTED]'
