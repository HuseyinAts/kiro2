def calculate_cosine_similarity(
    self,
    embedding1: np.ndarray,
    embedding2: np.ndarray
) -> float:
    """Calculate cosine similarity between two embeddings"""
    dot_product = np.dot(embedding1, embedding2)
    norm1 = np.linalg.norm(embedding1)
    norm2 = np.linalg.norm(embedding2)

    if norm1 == 0 or norm2 == 0:
        return 0.0

    return dot_product / (norm1 * norm2)

KANIT 3: Fisher Information Algoritması (Gerçek IRT Matematiği)

def _calculate_fisher_information(
    self,
    theta: float,
    a: float,
    b: float,
    c: float = 0.25
) -> float:
    """
    Fisher Information for 3PL model
    I(θ) = a² * [(P - c)² / ((1 - c)² * P * Q)]
    """
    P = self._probability_correct_3pl(theta, a, b, c)
    Q = 1 - P

    if P == 0 or Q == 0:
        return 0.0

    numerator = (P - c) ** 2
    denominator = (1 - c) ** 2 * P * Q

    return (a ** 2) * (numerator / denominator)

KANIT 5: EAP Ability Estimation (Gerçek Bayesian Inference)

def _estimate_ability_eap(
    self,
    responses: List[Dict],
    prior_mean: float = 0.0,
    prior_std: float = 1.0
) -> Tuple[float, float]:
    """
    Expected A Posteriori (EAP) ability estimation
    Bayesian inference approach
    """
    # Numerical integration over theta range
    theta_range = np.linspace(-4, 4, 81)
    posterior = np.zeros_like(theta_range)

    # Calculate posterior distribution
    for i, theta in enumerate(theta_range):
        likelihood = 1.0

        for response in responses:
            prob = self._probability_correct_3pl(
                theta,
                response["irt_a"],
                response["irt_b"],
                response.get("irt_c", 0.25)
            )

            if response["is_correct"]:
                likelihood *= prob
            else:
                likelihood *= (1 - prob)

        # Prior: Normal(0, 1)
        prior = np.exp(-(theta - prior_mean)**2 / (2 * prior_std**2))
        posterior[i] = likelihood * prior

    # Normalize
    posterior /= np.sum(posterior)

    # Calculate EAP (expected value)
    eap_theta = np.sum(theta_range * posterior)

    # Calculate standard error
    variance = np.sum((theta_range - eap_theta)**2 * posterior)
    sem = np.sqrt(variance)

    return eap_theta, sem

def submit_review(self, submission: ReviewSubmission) -> Dict:
    """
    Expert submits review
    INNOVATION: AI feedback loop for continuous learning
    """
    task = next(
        (t for t in self.task_queue if t.task_id == submission.task_id),
        None
    )

    expert = self.experts.get(submission.expert_id)

    # Update task
    task.completed_time = datetime.now()
    self.task_queue.remove(task)
    self.completed_tasks.append(task)

    # Update expert metrics
    expert.total_reviews += 1
    expert.points += self._calculate_review_points(submission, task)

    # Update approval rate
    if submission.decision == ReviewDecision.APPROVE:
        expert.approval_rate = (
            (expert.approval_rate * (expert.total_reviews - 1) + 1.0)
            / expert.total_reviews
        )

    # Award badges
    self._check_and_award_badges(expert)

    # Update leaderboard
    self._update_leaderboard()

def _find_best_expert(
    self,
    task: ReviewTask,
    available_experts: List[ExpertProfile]
) -> Optional[str]:
    """
    Smart expert assignment based on expertise matching
    """
    if not available_experts:
        return None

    # Calculate expertise match scores
    expert_scores = []
    for expert in available_experts:
        # Specialization match
        spec_match = 1.0 if task.konu in expert.specializations else 0.3

        # Load balance (prefer less busy experts)
        assigned_count = len([t for t in self.task_queue if t.assigned_expert_id == expert.user_id])
        load_penalty = max(0, 1.0 - (assigned_count * 0.1))

        # Quality score bonus
        quality_bonus = expert.quality_score / 100.0

        # Combined score
        score = spec_match * load_penalty * quality_bonus

        expert_scores.append((expert.user_id, score))

    # Select best expert
    best_expert = max(expert_scores, key=lambda x: x[1])
    return best_expert[0]
